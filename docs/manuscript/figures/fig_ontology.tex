                 text width=30mm, minimum height=6mm},

  outcome/.style={box, text width=25mm},   % not `out`: /tikz/out is reserved by to[out=..]

\node[box] (chg) {Environment change at $\tau^{*}$};
\node[box, below=of chg] (res)
  {Error excess $\Delta e$ on the stream $e_t$\\[1pt]\emph{endogenous residual}};

\node[loop, below left=6mm and -13mm of res] (adapt)
  {Internal correction\\replaces trees};
\node[box,  below right=6mm and -13mm of res] (mon)
  {External monitor\\accumulates toward $\lambda$};

\node[box, below=13mm of res, text width=44mm] (cmp)
  {Window $\tau_{\mathrm{erase}}$ \emph{vs.} required accumulation
   $T(\lambda,\Delta e)$};

\node[bad,  below left=6mm and -6mm of cmp] (blind)
  {$\tau_{\mathrm{erase}} < T$\\\textbf{starvation}\\missed detection
   $\Rightarrow$ \textbf{blind spot}};
\node[outcome, below right=6mm and -6mm of cmp] (late)
  {$\tau_{\mathrm{erase}} \geq T$\\alarm fires; if delayed,\\after recovery};

\node[outcome, below=6mm of blind, text width=34mm, xshift=6mm] (flood)
  {Lower $\lambda$ to fit the window\\$\Rightarrow$ alarm flooding under
   volatility};

\node[outcome, right=3mm of res, text width=21mm, xshift=6mm] (px)
  {Monitor on $P(X)$:\\outside the loop};

\draw[ar] (chg) -- (res);
\draw[ar] (res.south west) -- (adapt.north);
\draw[ar] (res.south east) -- (mon.north);
\draw[fb]  (adapt.west) -- ++(-4mm,0) |- (res.west)
  node[pos=0.25, rotate=90, anchor=south, font=\tiny] {negative feedback};
\draw[ar] (adapt.south) -- (cmp.north west);
\draw[ar] (mon.south)   -- (cmp.north east);
\draw[ar] (cmp.south west) -- (blind.north);
\draw[ar] (cmp.south east) -- (late.north);
\draw[ar] (blind.south) -- (flood.north);
\draw[ar] (res.east) -- (px.west);

\end{tikzpicture}
\caption{Relations between the retained effects. The classifier acts by negative
feedback on the very stream the monitor reads, bounding the observation window at
$\tau_{\mathrm{erase}}$. Recall collapse (left) and precision collapse (bottom)
are two settings of one threshold $\lambda$, not two phenomena. A monitor on the
input distribution is not exposed to the loop, at the cost of blindness to a
change in $P(Y\mid X)$ at fixed $P(X)$.}
\label{fig:ontology}
